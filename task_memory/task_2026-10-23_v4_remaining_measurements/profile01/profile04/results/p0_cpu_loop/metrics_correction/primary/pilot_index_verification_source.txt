"""Validate the completed index and preserve the failed, once-only P0 pilot."""
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path
import gzip
import hashlib
import json
import sys
import time

REPO = Path('/data/ycfeng/Frontier/.worktrees/glm53-dsv4-vllm028')
RECORDS = REPO / 'task_memory/task_2026-10-23_v4_remaining_measurements/profile01/profile04/results/p0_cpu_loop/metrics_correction/primary'
PREFIX = Path('/data/ycfeng/tmp/glm53_dsv4/p0_metrics_affected_20261025_01/prefill_dp0_rank0')
RECEIPT = json.loads(PREFIX.with_suffix('.receipt.json').read_text())
CHECKS = []
STARTED = time.monotonic()


def check(name, value):
    CHECKS.append({'name': name, 'pass': bool(value)})
    assert value, name


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


check('pilot_exit124', RECEIPT['returncode'] == 124)
check('final_json_absent', not PREFIX.with_suffix('.json').exists())
check('actual30GiB', RECEIPT['actual_cgroup']['limits']['memory.limit_in_bytes'] == '32212254720')
check('adapter_byte_binding', sha(REPO / 'tests/analysis/v4_trace_audit_metrics.py') == RECEIPT['adapter_sha256'])
for row in RECEIPT['outputs']:
    check('closed_output:' + Path(row['path']).name, sha(row['path']) == row['sha256'])
index = PREFIX.with_suffix('.kernels.jsonl.gz')
rows, last, counts, digest = 0, -1, Counter(), hashlib.sha256()
with gzip.open(index, 'rb') as stream:
    for line in stream:
        digest.update(line)
        value = json.loads(line)
        assert len(value) == 6 and value[0] > last
        last = value[0]
        counts[value[5]] += 1
        rows += 1
stages = [json.loads(line) for line in PREFIX.with_suffix('.obs.jsonl').read_text().splitlines()]
finished = next(row for row in stages if row['stage'] == 'kernel_pass_done')
check('complete_index_status_counts', dict(counts) == finished['status_rows'])
check('complete_index_count', rows == 1215919)
old_index = Path('/data/ycfeng/tmp/glm53_dsv4/p0_gc_observation_20261024/p0_load_gc_01.kernels.jsonl.gz')
old_receipt = json.loads(old_index.with_name('p0_load_gc_01.receipt.json').read_text())
check('retained_loading_gc_index_sha', sha(old_index) == old_receipt['files'][str(old_index)])
old_digest, old_lines = hashlib.sha256(), 0
with gzip.open(old_index, 'rb') as stream:
    for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
        old_digest.update(block)
        old_lines += block.count(b'\n')
check('kernel_index_exact_decompressed_equality', digest.hexdigest() == old_digest.hexdigest() and old_lines == rows)
result = {'status': 'FAILED_P0_PILOT_NO_FINAL_JSON', 'observed_utc': datetime.now(timezone.utc).isoformat(),
          'checks': CHECKS, 'python': sys.version, 'seconds': time.monotonic() - STARTED,
          'returncode': 124, 'affected_invocations_used': 1, 'remaining_invocations_started': 0,
          'conditionally_remaining_ranks': 7, 'actual_limit_bytes': 32212254720,
          'complete_index_rows': rows, 'complete_index_sha256': sha(index),
          'decompressed_index_sha256': digest.hexdigest(), 'index_equals_retained_loading_gc_index': True,
          'status_counts': dict(counts), 'last_stage': finished,
          'retained_index': {'path': str(old_index), 'sha256': sha(old_index),
                             'receipt_sha256': sha(old_index.with_name('p0_load_gc_01.receipt.json'))},
          'stages': {row['stage']: row['t'] for row in stages if not row['stage'].endswith('progress')},
          'max_observed_sparse_rss_mib': max(row['rss_mb'] for row in stages),
          'missing_final_stages': [name for name in ('per_stream_done', 'sync_done', 'steps_done', 'result_built',
                                                   'output_written', 'done', 'process_end')
                                   if not any(row['stage'] == name for row in stages)],
          'full_trace_parses': 0, 'full_adapter_invocations': 0, 'gpu_operations': 0,
          'limit': 'Index completion and the CPU failure only; no final metrics/useful steps, R5/full32, ownership, dependency, cost or reference acceptance.'}
with (RECORDS / 'pilot_failure.json').open('x') as stream:
    json.dump(result, stream, indent=2)
    stream.write('\n')
print(json.dumps({'status': result['status'], 'checks': len(CHECKS), 'index_rows': rows,
                  'index_exact': True, 'seconds': result['seconds']}))
