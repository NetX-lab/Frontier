def deeply_immutable(value):
    if value is None or type(value) in (str, int, float, bool):
        return True
    if type(value) is tuple:
        return all(deeply_immutable(item) for item in value)
    if type(value) in IMMUTABLE_CLASSES:
        return all(deeply_immutable(item) for item in vars(value).values())
    return False


def immutable_snapshot(self, memo):
    """Keep mutable children deeply copied; only verified immutable values reuse identity."""
    if deeply_immutable(self):
        return self
    result = object.__new__(type(self))
    memo[id(self)] = result
    vars(result).update(copy.deepcopy(vars(self), memo))
    return result

def walk(value, path):
    if dataclasses.is_dataclass(value):
        cls = type(value)
        classes[cls.__module__ + '.' + cls.__name__] += 1
        domain = domains.setdefault(cls.__module__ + '.' + cls.__name__, {})
        for field in dataclasses.fields(value):
            domain.setdefault(field.name, set()).add(type(getattr(value, field.name)).__name__)
        if cls.__dataclass_params__.frozen and not deeply_immutable(value):
            frozen_mutable_paths.append(path)
        for name, item in vars(value).items():
            walk(item, path + '.' + name)
    elif isinstance(value, dict):
        for name, item in value.items():
            walk(item, path + '.' + str(name))
    elif isinstance(value, (list, tuple)):
        for index, item in enumerate(value):
            walk(item, path + '[' + str(index) + ']')
